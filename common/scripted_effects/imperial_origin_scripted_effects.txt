io_set_planet_pop = {
    set_variable = {
        which = pop_to_add
        value = $pop$
    }
    export_trigger_value_to_variable = {
        trigger = count_owned_pop_amount
        variable = pop_current
    }
    subtract_variable = {
        which = pop_to_add
        value = pop_current
    }
    divide_variable = {
        which = pop_to_add
        value = 100
    }
    floor_variable = pop_to_add
    while = {
        count = pop_to_add
        if = {
            limit = {
                OR = {
                    has_origin = origin_syncretic_evolution
                    has_origin = origin_necrophage
                }
            }
            create_pop_group = {
                species = owner_main_species
                size = 75
            }
            create_pop_group = {
                species = event_target:syncretic_species
                size = 25
            }
        }
        else_if = {
            limit = {
                has_origin = origin_mechanists
            }
            create_pop_group = {
                species = owner_main_species
                size = 80
            }
            create_pop_group = {
                species = event_target:mechanical_species
                size = 20
            }
        }
        else_if = {
            limit = {
                OR = {
                    has_civic = civic_machine_assimilator
                    has_civic = civic_hive_bodysnatcher
                }
            }
            create_pop_group = {
                species = owner_main_species
                size = 80
            }
            create_pop_group = {
                species = event_target:secondery_species
                size = 20
            }
        }
        else = {

                species = owner_main_species
                size = 100
