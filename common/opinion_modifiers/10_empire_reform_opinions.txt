	trigger = {
		OR = {
			is_galactic_emperor = yes
		}
	}
	opinion = {
		modifier = {
			add = -1000
			is_galactic_emperor = yes
			FROM = { 
			is_federation_leader = yes }
			is_same_species = FROM
		}
	}
}

opinion_chosen_of_the_shroud = {
	trigger = {
		is_country_type = default
		
	}
	opinion = {
		modifier = {
			add = -1000
			is_country_type = default
			NOR = {
				has_origin = origin_chosen_of_the_shroud
				AND = {
					has_any_overlord = yes
					overlord = {
						has_origin = origin_chosen_of_the_shroud
					}
				}
			}
			FROM = { has_origin = origin_chosen_of_the_shroud }
		}
		modifier = {
			add = -1000
			has_origin = origin_chosen_of_the_shroud
			FROM = {
				NOR = {
					has_origin = origin_chosen_of_the_shroud
					AND = {
						has_any_overlord = yes
						overlord = {
							has_origin = origin_chosen_of_the_shroud
						}
					}
				}
				is_country_type = default
			}
		}
	}
}
