cb_io_chosen_of_the_shroud = {
	show_in_diplomacy = yes
	show_notification = yes
	potential = {
		OR = {
			has_origin = origin_chosen_of_the_shroud
			is_galactic_custodian = yes
			is_galactic_emperor = yes
		}
	}
	is_valid = {
		OR = {
			AND = {
				has_country_flag = composer_covenant_confirmed
				from = { NOT = { has_country_flag = composer_covenant_confirmed } }
			}
			AND = {
				has_country_flag = eater_covenant_confirmed
				from = { NOT = { has_country_flag = eater_covenant_confirmed } }
			}
			AND = {
				has_country_flag = instrument_covenant_confirmed
				from = { NOT = { has_country_flag = instrument_covenant_confirmed } }
			}
			AND = {
				has_country_flag = whisperers_covenant_confirmed
				from = { NOT = { has_country_flag = whisperers_covenant_confirmed } }
			}
			AND = {
				OR = {
					is_galactic_custodian = yes
					is_galactic_emperor = yes
				}
				from = { has_origin = origin_chosen_of_the_shroud }
			}
		}
		NOT = { is_in_federation_with = from }
		NOT = { is_overlord_to = FROM }
		is_subject = no
	}
}
