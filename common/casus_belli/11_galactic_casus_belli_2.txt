cb_io_intervention = { #干预战争借口，建立帝国总督府或共同体监管国
	show_in_diplomacy = yes
	show_notification = yes
	potential = {
		OR = {
		is_galactic_emperor = yes
		is_galactic_custodian = yes
		}
	}
	is_valid = {
			from = {
				is_galactic_community_member = yes
				OR = {	#140权威以上，帝国成员内战
					AND = {
						has_galactic_emperor = yes
						imperial_authority > 140
						is_at_war = yes
						any_war = {
							is_war_participant = { who = prev side = attackers }
							any_defender = { is_galactic_community_member = yes }
						}
					}
					has_modifier = resolution_denounced #被帝国议会/理事会谴责
					is_in_breach_of_any = yes #违反银河法查明是义军
					NOT = {
						is_in_federation_with = root
					}
				}
			}
	}
}
