event = {
	id = galactic_starts.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_timed_global_flag = {
			flag = imperium_origins_active
			days = 30
		}
	if = {
		limit = {
			any_country = {
			has_origin = origin_galactic_hegemon
			}
		}
		every_playable_country = {
			limit = {
				OR = {
				has_origin = origin_common_ground

				# has_origin = originsextended_research
				}
			}
			country_event = { id = galactic_expand.2 }
		}
		every_playable_country = {
			every_playable_country = {
				establish_communications_no_message = prev
			}
		} 
		set_global_flag = galactic_community_founded
		set_global_flag = galactic_council_formed
		
		every_playable_country = {
			if = {
				limit = {
					OR = {
						has_origin = origin_common_ground
						has_origin = origin_common_ground_npc
						has_origin = origin_hegemon
						has_origin = origin_hegemon_npc
						has_origin = origin_scion
						has_civic = civic_fanatic_purifiers
						has_civic = civic_hive_devouring_swarm
						has_civic = civic_machine_terminator
						has_civic = civic_outside_the_border
						# has_origin = originsextended_outreach
						# has_origin = originsextended_outreach_npc
						# has_origin = originsextended_research
						# has_origin = originsextended_research_npc
						# has_origin = origin_outside_the_border
						# has_origin = origin_secondary_hegemon
						has_origin = origin_secondary_hegemon_vassal
						has_origin = origin_imperial_vassal
						has_origin = origin_imperial_vassal_overlord
						# has_origin = origin_red_satellite
						# has_origin = origin_red_satellite_overlord
						# has_origin = origin_ofe_fallen_empire
						has_origin = origin_chosen_of_the_shroud
						has_origin = origin_broken_shackles
						has_origin = origin_payback
						has_origin = origin_fear_of_the_dark
						AND = {
							has_origin = origin_internal_border
							NOT = { has_country_flag = internal_border_community_state }
						}
					}
				}
				set_country_flag = galcom_form_nay
			}
			else = {
				add_to_galactic_community_no_message = yes
			}
		}
		random_playable_country = {
			limit = {
				has_origin = origin_galactic_hegemon
		}
		country_event = { id = galactic_origin.1 }	
		pass_resolution_no_cooldown = resolution_council_denouncement
		set_variable = {
			which = galaxy_food
			value = 0
		}
		set_variable = {
			which = galaxy_minerals
			value = 0
		}
		set_variable = {
			which = galaxy_energy
			value = 0
		}
		set_variable = {
			which = galaxy_consumer_goods
			value = 0
		}
		set_variable = {
			which = galaxy_alloys
			value = 0
		}
		}
		every_playable_country = {
			limit = {
				has_origin = origin_clone_army
				}
			country_event = { id = clones.22 }
		}
		#划分领土
			random_playable_country = {
				limit = { is_ai = no NOT = { has_country_flag = tech_setting } }
				set_variable = {
					which = start_tech
					value = 3
				}
				set_variable = {
					which = start_a
					value = 1
				}
				set_variable = {
					which = start_b
					value = 2
				}
				set_variable = {
					which = start_c
					value = 1
				}
				set_variable = {
					which = start_f
					value = 1
				}
				set_variable = {
					which = start_empire
					value = 1
				}
				if = {
					limit = {
						any_country = {
							is_galactic_emperor = yes
							has_civic = civic_vilified
						}
					}
					set_variable = {
						which = start_empire
						value = 0
					}
				}
				set_variable = {
					which = start_h
					value = 2
				}
				set_variable = {
					which = start_i
					value = 1
				}
				set_variable = {
					which = start_j
					value = 1
				}
				set_variable = {
					which = start_expand
					value = 9
				}
				country_event = { id = galactic_origin.100 }
				set_country_flag = tech_setting
			}
		}
	}
}
