planet_event = {

			set_species_flag = syncretic_species@root.owner

					limit = {
						root.owner = {

								is_authoritarian = yes
								is_xenophobe = yes
							}
						}
					}
					set_citizenship_type = {
						country = root.owner
						type = citizenship_slavery
					}
				}
			}
		owner = {
		io_save_nation_pop = yes
		give_technology = {
			tech = tech_curator_lab
			message = no
		}
		if = {
			limit = { has_ancrel = yes }
			give_technology = {
				tech = tech_archeology_lab_ancrel
				message = no
			}
		}
		if = {
			limit = { has_ancrel = no }
			give_technology = {
				tech = tech_archeology_lab
				message = no
			}
		}
		give_technology = {
			tech = tech_mine_betharian
			message = no
		}
		give_technology = {
			tech = tech_mine_volatile_motes
			message = no
		}
		give_technology = {
			tech = tech_mine_exotic_gases
			message = no
		}
		give_technology = {
			tech = tech_mine_rare_crystals
			message = no
		}
		give_technology = {
			tech = tech_mine_zro
			message = no
		}
		give_technology = {
			tech = tech_mine_dark_matter
			message = no
		}
		give_technology = {
			tech = tech_mine_living_metal
			message = no
		}
		capital_scope = {
			io_upgrade_planet_3 = yes
			add_district_and_planet_size_if_needed_effect = {
				district = district_industrial
			}
		}
		random_planet_within_border = {
			limit = { has_planet_flag = research_habitat }
			set_owner = root
			set_planet_entity = {
				entity = habitat_phase_02_entity
				graphical_culture = root
			}
			set_planet_size = 6
			set_planet_flag = advanced_habitat
