    id = io_start_events.100

    hide_window = yes

    is_triggered_only = yes

    trigger = {
        NOR = {
            has_origin = origin_void_dwellers
            has_origin = origin_void_machines
            has_origin = origin_fear_of_the_dark
            has_origin = origin_unplugged
            is_low_tech_start = yes
            is_nomadic = yes
        }
    }

    immediate = {
        #给予资源
        io_add_start_resources = yes
        #母星设置
        capital_scope = {
            io_save_secondery_pop = yes
            io_clear_planet = yes
            clear_blockers = yes
            if = {
                limit = {
                    owner = {
                        has_country_flag = io_fe_emperor
                    }
                }
                io_fe_planet_set = yes
            }
            else_if = {
                limit = {
                    owner = {
                        is_galactic_emperor = yes
                        NOT = {
                            check_variable = {
                                which = io_setting_empire
                                value = 0
                            }
                        }
                    }
                }
                io_districts_city_world = yes
                io_building_captial = yes
            }
            else = {
                io_districts_captial = yes
                io_districts_placement = yes
                io_building_captial = yes
            }
            if = {
                limit = {
                    owner = {
                        is_galactic_emperor = yes
                    }
                }
                if = {
                    limit = {
                        owner = {
                            check_variable = {
                                which = io_setting_empire
                                value >= 4
                            }
                        }
                    }
                    io_set_planet_pop = {
                        pop = 20000
                    }
                }
                if = {
                    limit = {
                        owner = {
                            check_variable = {
                                which = io_setting_empire
                                value = 3
                            }
                        }
                    }
                    io_set_planet_pop = {
                        pop = 20000
                    }
                }
                else_if = {
                    limit = {
                        owner = {
                            check_variable = {
                                which = io_setting_empire
                                value >= 2
                            }
                        }
                    }
                    io_set_planet_pop = {
                        pop = 15000
                    }
                }
                else = {
                    io_set_planet_pop = {
                        pop = 10000
                    }
                }

                                is_same_value = root
                            }
                        }

                        set_starbase_size = starbase_waystation_2
